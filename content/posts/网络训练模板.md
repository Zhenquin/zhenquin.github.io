### 介绍
这里以经典的alexnet网络为例，展示如何使用pytorch实现一个简单的网络训练模板
### 具体实现
- 导入必要的库：
```python
import torch
import torchvision
from torch import nn
from torchvision import transforms
import matplotlib.pyplot as plt
```
- 定义alexnet网络：
```python
class AlexNet(nn.Module):
    def __init__(self):
        super(AlexNet, self).__init__()
        self.conv1 = nn.Conv2d(3, 96, kernel_size=11, stride=4, padding=2)
        self.pool1 = nn.MaxPool2d(kernel_size=3, stride=2)
        self.conv2 = nn.Conv2d(96, 256, kernel_size=5, stride=1, padding=2)
        self.pool2 = nn.MaxPool2d(kernel_size=3, stride=2)
        self.conv3 = nn.Conv2d(256, 384, kernel_size=3, stride=1, padding=1)
        self.conv4 = nn.Conv2d(384, 384, kernel_size=3, stride=1, padding=1)
        self.conv5 = nn.Conv2d(384, 256, kernel_size=3, stride=1, padding=1)
        self.pool3 = nn.MaxPool2d(kernel_size=3, stride=2)
        self.fc1 = nn.Linear(256 * 6 * 6, 4096)
        self.fc2 = nn.Linear(4096, 4096)
        self.fc3 = nn.Linear(4096, 10)
        self.flatten = nn.Flatten()
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.5)

    def forward(self, x):
        # 卷积层
        x = self.relu(self.conv1(x))
        x= self.pool1(x)
        x = self.relu(self.conv2(x))
        x = self.pool2(x)
        x = self.relu(self.conv3(x))
        x = self.relu(self.conv4(x))
        x = self.relu(self.conv5(x))
        x = self.pool3(x)
        # 展平全连接
        x = self.flatten(x)
        x = self.dropout(self.relu(self.fc1(x)))
        x = self.dropout(self.relu(self.fc2(x)))
        x = self.fc3(x)
        return x

```
- 配置训练超参数：
```python
config = {
    'batch_size': 64,
    'num_workers': 4,
    'learning_rate': 0.01,
    'save_path': 'best_model_of_AlexNet.pth',
    'num_epochs': 50,
    'device': 'cuda' if torch.cuda.is_available() else 'cpu',
}
```
- 数据预处理：
```python

model = AlexNet()
model = model.to(config['device'])

train_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.RandomHorizontalFlip(p=0.5),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
])

val_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
])

# 数据集加载
train_set = torchvision.datasets.CIFAR10(root='./data', train=True, download=False, transform=train_transform)
val_set = torchvision.datasets.CIFAR10(root='./data', train=False, download=False, transform=val_transform)

train_loader = torch.utils.data.DataLoader(train_set, batch_size=config['batch_size'], shuffle=True, num_workers=config['num_workers'],)
val_loader = torch.utils.data.DataLoader(val_set, batch_size=config['batch_size'], shuffle=False, num_workers=config['num_workers'])
```
- 配置优化器与损失函数：
```python
# 优化器与损失函数
optimizer = torch.optim.Adam(model.parameters(), lr=config['learning_rate'])
criterion = nn.CrossEntropyLoss()

```
- 训练循环：
```python
# 训练，评估，保存最佳结果
best_accuracy = 0.0
for epoch in range(config['num_epochs']):
    # 训练部分
    model.train()
    total_loss = 0
    for image_r, label_r in train_loader:
        # 数据传递
        image, label = image_r.to(config['device']), label_r.to(config['device'])

        # 前向传播
        output = model(image)
        loss = criterion(output, label)

        # 反向传播
        optimizer.zero_grad()
        loss.backward()
        # 更新参数
        optimizer.step()
        total_loss += loss.item()
    avg_train_loss = total_loss / len(train_loader)

    # 评估部分
    model.eval()
    correct = 0
    total = 0

    with torch.no_grad():  # 关闭梯度计算
        for image_r, label_r in val_loader:
            # 数据传递
            image, label = image_r.to(config['device']), label_r.to(config['device'])
            outputs = model(image)
            _, predicted = torch.max(outputs, 1)
            total += label.size(0)
            correct += (predicted == label).sum().item()

    val_accuracy = 100 * correct / total
    print(f"Val Accuracy: {val_accuracy:.2f}%")

    if val_accuracy > best_accuracy:
        best_accuracy = val_accuracy
        torch.save(model.state_dict(), config['save_path'])
        print(f"✅ 发现更好的模型，已保存！验证准确率: {val_accuracy:.2f}%")

    print(
        f"Epoch [{epoch + 1}/{config['num_epochs']}], Train Loss: {avg_train_loss:.4f}, Val Acc: {val_accuracy:.2f}%")

print(f"训练结束！最佳验证准确率为: {best_accuracy:.2f}%")
```
